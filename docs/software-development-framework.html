📏 1.1 PEP 8 Style Guide

PEP 8 is the official Python style guide. Consistent formatting makes code easier to read, review, and maintain. Tools like Black and flake8 can enforce these rules automatically.

Indentation & Line Length

Python 
# ✅ Use 4 spaces (never tabs)
def calculate_total(items: list) -> float:
    total = 0.0
    for item in items:
        if item.price > 0:
            total += item.price * item.quantity
    return total

# ✅ Max line length: 79 chars (PEP 8) or 88 chars (Black)
# ✅ Use implicit line continuation inside brackets
result = some_function(
    argument_one,
    argument_two,
    argument_three,
)

# ✅ Long strings: use parentheses
message = (
    "This is a very long string that would exceed the line length"
    " limit if written on a single line."

📖 1.4 Documentation Standards

Good documentation is essential for maintainability. Python supports three popular docstring styles. Choose one and stick with it throughout your project.

Google Style Docstring (Recommended)

Python 
def calculate_compound_interest(
    principal: float,
    rate: float,
    periods: int,
    compounds_per_period: int = 12,
) -> float:
    """Calculate compound interest for an investment.

    Uses the standard compound interest formula:
    A = P(1 + r/n)^(nt)

    Args:
        principal: The initial investment amount in dollars.
        rate: Annual interest rate as a decimal (e.g., 0.05 for 5%).
        periods: Number of years to compound.
        compounds_per_period: Number of times interest compounds per year.
            Defaults to 12 (monthly).

    Returns:
        The total amount after compound interest, as a float.

    Raises:
        ValueError: If principal or rate is negative.
        TypeError: If periods is not an integer.

    Example:
        >>> calculate_compound_interest(1000, 0.05, 10)
        1647.0094804784497
        >>> calculate_compound_interest(5000, 0.03, 5, 4)
        5803.7748776279015
    """
    if principal < 0 or rate < 0:
        raise ValueError("Principal and rate must be non-negative")
    return principal * (1 + rate / compounds_per_period) ** (
        compounds_per_period * periods
    )

NumPy Style Docstring

def normalize_data(data: list[float]) -> list[float]:
    """Normalize data to range [0, 1].

    Parameters
    ----------
    data : list of float
        The input data to normalize. Must contain at least 2 values
        and all values must be finite.

    Returns
    -------
    list of float
        Normalized data where each value is in [0, 1].

    Raises
    ------
    ValueError
        If data is empty or contains only one unique value.

    Notes
    -----
    Uses min-max normalization: (x - min) / (max - min)

    Examples
    --------
    >>> normalize_data([0, 5, 10])
    [0.0, 0.5, 1.0]
    """
    min_val = min(data)
    max_val = max(data)
    if min_val == max_val:
        raise ValueError("Cannot normalize data with single unique value")
    return [(x - min_val) / (max_val - min_val) for x in data]

🔤 1.5 Type Hints & Annotations

Type hints improve code readability, enable IDE autocompletion, and allow static analysis with tools like mypy. As of Python 3.10+, many typing module types can be used natively.

Python 
from __future__ import annotations
from typing import (
    Any, Callable, Dict, Generic, List, Optional,
    Protocol, Tuple, TypeVar, Union,
)
from collections.abc import Sequence, Iterator

# TypeVar for generic functions
T = TypeVar("T")
K = TypeVar("K")
V = TypeVar("V")

# Basic type hints
def greet(name: str, age: int) -> str:
    return f"Hello {name}, you are {age} years old"

# Optional (value or None)
def find_user(user_id: int) -> Optional["User"]:
    return None  # May return None

# Union (Python 3.10+: int | str)
def process(value: Union[int, str]) -> str:
    return str(value)

# Collections
def get_scores() -> Dict[str, List[int]]:
    return {"alice": [95, 87, 92], "bob": [78, 83]}

# Callable
def apply(func: Callable[[int], str], value: int) -> str:
    return func(value)

# Tuple (fixed length)
def get_coordinates() -> Tuple[float, float]:
    return (51.5, -0.1)

# Generic class
class Repository(Generic[T]):
    def __init__(self) -> None:
        self._items: List[T] = []

    def add(self, item: T) -> None:
        self._items.append(item)

    def get_all(self) -> List[T]:
        return self._items.copy()

# Protocol (structural subtyping)
class Drawable(Protocol):
    def draw(self) -> None: ...

# Python 3.10+ union syntax
def modern_func(x: int | str | None = None) -> str:
    return str(x) if x is not None else ""
⚠️ 1.6 Error Handling

try/except/else/finally Pattern

Python 
import logging

logger = logging.getLogger(__name__)


def read_user_data(file_path: str) -> dict:
    """Read and parse user data from a JSON file."""
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            data = json.load(f)
    except FileNotFoundError:
        logger.error("File not found: %s", file_path)
        raise  # Re-raise for caller to handle
    except json.JSONDecodeError as e:
        logger.error("Invalid JSON in %s: %s", file_path, e)
        raise ValueError(f"File contains invalid JSON") from e  # Exception chaining
    except PermissionError:
        logger.critical("Permission denied: %s", file_path)
        raise
    else:
        # Only runs if no exception occurred
        logger.info("Successfully loaded %d records", len(data))
        return data
    finally:
        # Always runs, even if exception is raised
        logger.debug("Completed file read attempt for %s", file_path)


# Custom exception hierarchy
class AppError(Exception):
    """Base exception for all application errors."""
    def __init__(self, message: str, code: str = "UNKNOWN") -> None:
        super().__init__(message)
        self.code = code


class ValidationError(AppError):
    """Raised when input validation fails."""
    def __init__(self, field: str, message: str) -> None:
        super().__init__(f"Validation failed for '{field}': {message}", "VALIDATION_ERROR")
        self.field = field


class NotFoundError(AppError):
    """Raised when a requested resource is not found."""
    def __init__(self, resource: str, resource_id: Any) -> None:
        super().__init__(
            f"{resource} with id '{resource_id}' was not found",
            "NOT_FOUND"
        )


# Usage with exception chaining
def get_user(user_id: int) -> "User":
    try:
        return db.query_user(user_id)
    except DatabaseError as e:
        raise NotFoundError("User", user_id) from e  # Preserves original traceback
⚠️
Warning
Never use bare except: without a specific exception type — it catches SystemExit, KeyboardInterrupt, and other critical exceptions you should not suppress. Always catch specific exception types.

🏛️ 2.1 SOLID Principles

SOLID is an acronym for five design principles that make software designs more understandable, flexible, and maintainable.

Figure 2.1 — SOLID Principles Overview

S — Single Responsibility Principle

❌ Violation
class UserManager:
    # Does TOO MANY things!
    def create_user(self, data): ...
    def send_welcome_email(self): ...
    def save_to_database(self): ...
    def log_activity(self): ...
    def generate_report(self): ...
✅ Correct
class UserRepository:
    def save(self, user): ...

class EmailService:
    def send_welcome(self, user): ...

class ActivityLogger:
    def log(self, event): ...

class UserService:
    # Orchestrates only
    def create_user(self, data):
        user = User(**data)
        self.repo.save(user)
        self.email.send_welcome(user)

O — Open/Closed Principle

Python 
from abc import ABC, abstractmethod

# Open for extension, closed for modification
class Discount(ABC):
    @abstractmethod
    def apply(self, price: float) -> float: ...

class PercentageDiscount(Discount):
    def __init__(self, percent: float) -> None:
        self.percent = percent
    def apply(self, price: float) -> float:
        return price * (1 - self.percent / 100)

class FixedDiscount(Discount):
    def __init__(self, amount: float) -> None:
        self.amount = amount
    def apply(self, price: float) -> float:
        return max(0, price - self.amount)

# Add new discounts WITHOUT modifying existing code
class BulkDiscount(Discount):
    def apply(self, price: float) -> float:
        return price * 0.85

def calculate_price(price: float, discount: Discount) -> float:
    return discount.apply(price)  # Works with ANY discount type

D — Dependency Inversion Principle

Python 
from abc import ABC, abstractmethod

# Abstract interface (abstraction)
class NotificationService(ABC):
    @abstractmethod
    def send(self, message: str, recipient: str) -> None: ...

# Concrete implementations
class EmailNotification(NotificationService):
    def send(self, message: str, recipient: str) -> None:
        print(f"Email to {recipient}: {message}")

class SlackNotification(NotificationService):
    def send(self, message: str, recipient: str) -> None:
        print(f"Slack to {recipient}: {message}")

# High-level module depends on abstraction, NOT concrete class
class OrderService:
    def __init__(self, notifier: NotificationService) -> None:
        self._notifier = notifier  # Injected dependency

    def place_order(self, order: "Order") -> None:
        # process order...
        self._notifier.send(f"Order {order.id} confirmed", order.email)

# Switch implementations without changing OrderService
service = OrderService(SlackNotification())
service = OrderService(EmailNotification())
🧩 2.2 Design Patterns

Singleton Pattern (Thread-Safe)

Python 
import threading
from typing import Any

class SingletonMeta(type):
    """Thread-safe Singleton metaclass."""
    _instances: dict[type, Any] = {}
    _lock: threading.Lock = threading.Lock()

    def __call__(cls, *args, **kwargs):
        with cls._lock:
            if cls not in cls._instances:
                instance = super().__call__(*args, **kwargs)
                cls._instances[cls] = instance
        return cls._instances[cls]


class DatabaseConnection(metaclass=SingletonMeta):
    def __init__(self, url: str = "") -> None:
        self.url = url
        self._connection = None

# Usage
db1 = DatabaseConnection("postgresql://localhost/mydb")
db2 = DatabaseConnection()
assert db1 is db2  # True — same instance

Factory Pattern

Python 
from abc import ABC, abstractmethod

class StorageBackend(ABC):
    @abstractmethod
    def upload(self, filename: str, data: bytes) -> str: ...
    @abstractmethod
    def download(self, url: str) -> bytes: ...

class S3Storage(StorageBackend):
    def upload(self, filename: str, data: bytes) -> str:
        return f"s3://bucket/{filename}"
    def download(self, url: str) -> bytes:
        return b""

class LocalStorage(StorageBackend):
    def upload(self, filename: str, data: bytes) -> str:
        return f"/uploads/{filename}"
    def download(self, url: str) -> bytes:
        return b""

def storage_factory(backend: str) -> StorageBackend:
    """Create the appropriate storage backend."""
    backends = {
        "s3": S3Storage,
        "local": LocalStorage,
    }
    if backend not in backends:
        raise ValueError(f"Unknown backend: {backend}")
    return backends[backend]()

# Usage
storage = storage_factory("s3")
url = storage.upload("report.pdf", data)

Observer Pattern

Python 
from typing import Callable, Any
from collections import defaultdict

class EventBus:
    """Simple publish/subscribe event system."""
    def __init__(self) -> None:
        self._handlers: dict[str, list[Callable]] = defaultdict(list)

    def subscribe(self, event: str, handler: Callable) -> None:
        self._handlers[event].append(handler)

    def publish(self, event: str, **data: Any) -> None:
        for handler in self._handlers[event]:
            handler(**data)

# Usage
bus = EventBus()
bus.subscribe("user.created", lambda **kw: print(f"Welcome {kw['name']}!"))
bus.subscribe("user.created", lambda **kw: logger.info("New user: %s", kw["name"]))
bus.publish("user.created", name="Alice", email="alice@example.com")

Strategy Pattern

Python 
from typing import Protocol

class SortStrategy(Protocol):
    def sort(self, data: list[int]) -> list[int]: ...

class QuickSort:
    def sort(self, data: list[int]) -> list[int]:
        return sorted(data)  # simplified

class MergeSort:
    def sort(self, data: list[int]) -> list[int]:
        return sorted(data)  # simplified

class Sorter:
    def __init__(self, strategy: SortStrategy) -> None:
        self.strategy = strategy

    def sort(self, data: list[int]) -> list[int]:
        return self.strategy.sort(data)

sorter = Sorter(QuickSort())
sorter.strategy = MergeSort()  # Swap at runtime

DRY — Don't Repeat Yourself

❌ Duplicated Logic
def validate_email_signup(email):
    if not email:
        raise ValueError("Email required")
    if "@" not in email:
        raise ValueError("Invalid email")

def validate_email_login(email):
    if not email:
        raise ValueError("Email required")
    if "@" not in email:
        raise ValueError("Invalid email")
✅ DRY Approach
def validate_email(email: str) -> None:
    """Reusable email validation."""
    if not email:
        raise ValueError("Email required")
    if "@" not in email:
        raise ValueError("Invalid email format")

def signup(email: str) -> None:
    validate_email(email)

def login(email: str) -> None:
    validate_email(email)

KISS — Keep It Simple, Stupid

❌ Over-engineered
def is_even(n: int) -> bool:
    return bool(
        int(
            bin(n).replace("0b", "")[-1]
        ) ^ 1
    )
✅ Simple & Clear
def is_even(n: int) -> bool:
    return n % 2 == 0

🚨
Critical Security Rule
NEVER hardcode secrets, passwords, API keys, or credentials in source code. Use environment variables or a secrets manager (AWS Secrets Manager, HashiCorp Vault, etc.).

Input Validation with Pydantic

Python 
from pydantic import BaseModel, EmailStr, Field, validator
from typing import Annotated

class CreateUserRequest(BaseModel):
    """Validated user creation request."""
    name: Annotated[str, Field(min_length=1, max_length=100)]
    email: EmailStr
    age: Annotated[int, Field(ge=0, le=150)]
    password: Annotated[str, Field(min_length=8)]

    @validator("password")
    def password_strength(cls, v: str) -> str:
        if not any(c.isupper() for c in v):
            raise ValueError("Password must contain an uppercase letter")
        return v

# Secrets management via environment variables
import os

# NEVER do this:
# DATABASE_URL = "postgresql://user:password123@localhost/mydb"  # ❌

# Always do this:
DATABASE_URL = os.environ["DATABASE_URL"]  # ✅ Raises if not set
SECRET_KEY = os.getenv("SECRET_KEY", "")   # ✅ Default empty

# SQL injection prevention — ALWAYS use parameterized queries
# NEVER: cursor.execute(f"SELECT * FROM users WHERE id = {user_id}")  ❌
cursor.execute("SELECT * FROM users WHERE id = %s", (user_id,))  # ✅

# Password hashing with bcrypt (never MD5 or SHA1!)
import bcrypt

def hash_password(password: str) -> bytes:
    salt = bcrypt.gensalt(rounds=12)
    return bcrypt.hashpw(password.encode(), salt)

def verify_password(password: str, hashed: bytes) -> bool:
    return bcrypt.checkpw(password.encode(), hashed)
⚡ 2.6 Performance Optimization

Generators vs Lists (Memory Efficiency)

Python 
import cProfile
import functools

# Generator — lazy evaluation, low memory
def read_large_file(filepath: str) -> Iterator[str]:
    """Memory-efficient file reading."""
    with open(filepath) as f:
        for line in f:
            yield line.strip()  # One line at a time, not all at once

# Generator expression vs list comprehension
total = sum(x ** 2 for x in range(1_000_000))  # ✅ Generator, ~O(1) memory
items = [x ** 2 for x in range(1_000_000)]    # ❌ List, ~8MB memory

# lru_cache for expensive repeated computations
@functools.lru_cache(maxsize=128)
def expensive_computation(n: int) -> int:
    """Fibonacci with memoization."""
    if n < 2:
        return n
    return expensive_computation(n - 1) + expensive_computation(n - 2)

# async/await for I/O-bound tasks
import asyncio
import aiohttp

async def fetch_urls(urls: list[str]) -> list[str]:
    """Fetch multiple URLs concurrently."""
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_one(session, url) for url in urls]
        return await asyncio.gather(*tasks)

# Profile code with cProfile
def profile_function() -> None:
    cProfile.run("my_slow_function()", sort="cumulative")
🔄
Generators
Use for large datasets. Lazy evaluation means only one item in memory at a time.
💾
lru_cache
Cache expensive function results. Great for pure functions called repeatedly with same args.
⚡
asyncio
Concurrent I/O without threads. Perfect for API calls, DB queries, file operations.
📊
cProfile
Measure before optimizing. Find the real bottleneck, not the assumed one.

🎵 3.2 What is Vibe Coding

Vibe Coding (coined by Andrej Karpathy in 2025) is a style of development where you describe what you want in natural language, let the AI generate the code, and iterate — trusting the AI to handle implementation details while you focus on intent and outcomes.

✅ When Vibe Coding Works Well

  • Rapid prototyping and MVPs
  • Boilerplate and scaffolding
  • Unfamiliar technologies or libraries
  • Data processing scripts
  • Converting specifications to code
  • Generating test data and fixtures

⚠️ When to Be Careful

  • Security-critical code paths
  • Complex business logic with edge cases
  • Performance-sensitive algorithms
  • Database schema migrations
  • Cryptographic implementations
  • Regulatory compliance code
🎯
Key Takeaway
Vibe coding doesn't mean abandoning engineering judgment — it means delegating implementation to AI while retaining responsibility for architecture, correctness, and security. You are still the engineer; the AI is a very capable pair programmer.

🎯 3.10 Best Practices for AI-Assisted Development
🔍Always review AI-generated code. Read every line before accepting. AI can produce plausible-looking but incorrect code.
🧪Test everything AI writes. Run existing tests after accepting suggestions. Write new tests to validate edge cases.
🔒Security-review AI code carefully. AI may produce vulnerable patterns (SQL injection, hardcoded secrets, insecure defaults).
📍Keep context focused. Smaller, specific prompts get better results than vague, sprawling requests.
🔄Iterate with feedback. If the first result isn't right, explain what's wrong and ask for a revision — don't start over.
📚Don't use AI as a documentation replacement. Verify library behavior against official docs; AI knowledge has a cutoff date.
👥Human ownership still matters. You are accountable for code you commit. "AI wrote it" is not an excuse for bugs or security issues.

Given I click the reset link in the email,
When the link is less than 24 hours old,

Given I click the reset link in the email,

## Status
Accepted

## Context
We need a relational database for our user data with strong ACID

## Decision
Use PostgreSQL 15+ as our primary database.

## Consequences
### Positive

- More operational overhead than SQLite

# RED: Write a failing test first
def test_calculate_discount_above_100_applies_10_percent():
    result = calculate_discount(price=150.0)
    assert result == 135.0  # Fails: function doesn't exist yet

# GREEN: Write minimum code to pass
def calculate_discount(price: float) -> float:
    if price > 100:
        return price * 0.9
    return price

# REFACTOR: Improve without changing behavior
DISCOUNT_THRESHOLD = 100.0
DISCOUNT_RATE = 0.10

def calculate_discount(price: float) -> float:
    """Apply 10% discount for orders over $100."""
    if price > DISCOUNT_THRESHOLD:
        return price * (1 - DISCOUNT_RATE)
    return price

🗄️ 5.14 Database Standards
SQL 
-- Table naming: snake_case, plural
CREATE TABLE user_accounts (
    -- PK: always 'id', UUID recommended
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- FK: table_name_id pattern
    org_id      UUID NOT NULL REFERENCES organizations(id),
    -- snake_case columns
    email       VARCHAR(255) NOT NULL UNIQUE,
    full_name   VARCHAR(200) NOT NULL,
    is_active   BOOLEAN NOT NULL DEFAULT TRUE,
    -- Always include audit timestamps
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    deleted_at  TIMESTAMPTZ  -- Soft delete
);

-- Index strategy: index foreign keys and query columns
CREATE INDEX idx_user_accounts_org_id ON user_accounts(org_id);
CREATE INDEX idx_user_accounts_email ON user_accounts(email);
⚠️
Migration Rule
Never modify existing migration files. Once a migration has been run in any environment, it is immutable. Always create a new migration file to make changes. Use Alembic for SQLAlchemy projects.
